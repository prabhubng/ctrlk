Implementation Guides

Complete integration examples for each framework and grid library. Copy-paste ready.

CtrlK is headless — it emits events, your app renders the UI. These guides show the complete setup for each stack: install, connect, render palette, handle field jump.

Install

terminalShell
# Core engine (required)
npm install @ctrlk/core

# Pick your grid adapter
npm install @ctrlk/devextreme    # DevExtreme v21+
npm install @ctrlk/ag-grid       # AG Grid v28-31+

# Pick your framework adapter (optional)
npm install @ctrlk/angular       # Compiled Angular library (14+)
npm install @ctrlk/react         # React hooks

Angular + DevExtreme

Service

The service wraps CtrlK in Angular's DI system and bridges events into RxJS Observables via NgZone. Install the compiled library:

terminalShell
npm install @ctrlk/core @ctrlk/angular @ctrlk/devextreme

Add the runtime to angular.json:

angular.jsonJSON
"scripts": [
  "node_modules/@ctrlk/core/dist/ctrlk.runtime.min.js"
]

Import the module in your app:

app.module.tsAngular
import { CtrlkModule } from '@ctrlk/angular';

@NgModule({
  imports: [CtrlkModule.forRoot({ palette: true, density: true })],
})

Palette Component

Uses your existing DevExtreme DxPopup. Matches your theme. No inline styles.

command-palette.component.tsAngular
import { Component, OnInit, OnDestroy } from '@angular/core';
import { Subscription } from 'rxjs';
import { CtrlkService, PaletteRequest } from '@ctrlk/angular';

@Component({
  selector: 'app-command-palette',
  template: `
    <dx-popup [visible]="visible" [width]="540" [maxHeight]="440"
              [showTitle]="false" [shading]="true"
              [closeOnOutsideClick]="true"
              (onHidden)="visible = false">
      <dx-text-box
        placeholder="Type a command..."
        (onValueChanged)="onSearch($event)"
        (onKeyDown)="onKeyDown($event)">
      </dx-text-box>
      <dx-list [dataSource]="results"
               (onItemClick)="onSelect($event)">
        <div *dxTemplate="let item of 'item'">
          <div style="display:flex;justify-content:space-between">
            <span>{{ item.title }}</span>
            <span style="font-size:11px;opacity:0.5;font-family:monospace"
                  *ngIf="item.shortcut">
              {{ item.shortcut }}
            </span>
          </div>
        </div>
      </dx-list>
    </dx-popup>
  `,
})
export class CommandPaletteComponent implements OnInit, OnDestroy {
  visible = false;
  results: any[] = [];
  private sub: Subscription;
  private request: PaletteRequest | null = null;

  constructor(private ctrlk: CtrlkService) {}

  ngOnInit() {
    this.sub = this.ctrlk.paletteRequested$.subscribe(req => {
      this.request = req;
      this.results = req.commands;
      this.visible = true;
    });
  }

  onSearch(e: any) {
    this.results = this.request?.search(e.value || '') || [];
  }

  onSelect(e: any) {
    this.request?.execute(e.itemData.id);
    this.visible = false;
  }

  onKeyDown(e: any) {
    if (e.event.key === 'Escape') this.visible = false;
  }

  ngOnDestroy() { this.sub?.unsubscribe(); }
}

Grid Component

Wire the DevExtreme adapter in ngAfterViewInit. The adapter auto-disconnects when a new grid connects or the service is destroyed.

deals-grid.component.tsAngular
import { Component, ViewChild, AfterViewInit, OnDestroy } from '@angular/core';
import { DxDataGridComponent } from 'devextreme-angular';
import { DevExtremeAdapter } from '@ctrlk/devextreme';
import { CtrlkService } from '@ctrlk/angular';

@Component({
  template: `<dx-data-grid #grid [dataSource]="deals" keyExpr="dealId">
    <!-- your columns -->
  </dx-data-grid>`
})
export class DealsGridComponent implements AfterViewInit, OnDestroy {
  @ViewChild('grid') grid: DxDataGridComponent;
  private disconnect: (() => void) | null = null;

  constructor(private ctrlk: CtrlkService) {}

  ngAfterViewInit() {
    // Connect grid — returns cleanup function
    this.disconnect = this.ctrlk.connectGrid(
      new DevExtremeAdapter(this.grid.instance, { keyExpr: 'dealId' })
    );

    // Register page-specific commands
    this.ctrlk.registerCommand({
      id: 'deals.refresh',
      title: 'Refresh Deals',
      shortcut: 'Ctrl+R',
      category: 'Deals',
      execute: () => this.grid.instance.refresh(),
    });
  }

  ngOnDestroy() {
    this.disconnect?.();  // Cleans up adapter event listeners
  }
}

Field Jump (Ctrl+G)

For detail pages with form fields (issuer detail, deal detail), use the field jump event with section-grouped search:

issuer-detail.component.tsAngular
import { CtrlkService, FieldJumpRequest } from '@ctrlk/angular';

@Component({ template: `
  <!-- Add data-ctrlk-field to your KV rows -->
  <div *ngFor="let field of fields"
       [attr.data-ctrlk-field]="field.id"
       [attr.data-ctrlk-label]="field.label"
       [attr.data-ctrlk-section]="field.section">
    <label>{{ field.label }}</label>
    <span>{{ field.value }}</span>
  </div>

  <!-- Your field jump popup -->
  <dx-popup [visible]="jumpOpen" [width]="500" [showTitle]="false"
            (onHidden)="jumpOpen = false">
    <dx-text-box placeholder="Search fields..." (onValueChanged)="onJumpSearch($event)"></dx-text-box>
    <div *ngFor="let section of jumpSections">
      <div class="section-header">{{ section.name }}</div>
      <dx-list [dataSource]="section.fields" (onItemClick)="onJumpSelect($event)"></dx-list>
    </div>
  </dx-popup>
` })
export class IssuerDetailComponent implements OnInit {
  jumpOpen = false;
  jumpSections: any[] = [];
  private jumpRequest: FieldJumpRequest | null = null;

  constructor(private ctrlk: CtrlkService) {}

  ngOnInit() {
    // Set section ordering for this page
    this.ctrlk.fields.setSectionOrder([
      'About Issuer', 'Characteristics', 'Ratings',
      'ESG Relevancy', 'Related Deals', 'Active Assets',
    ]);

    this.ctrlk.fieldJumpRequested$.subscribe(req => {
      this.jumpRequest = req;
      req.discover();  // Re-scan DOM for [data-ctrlk-field]
      const { sections } = req.searchGrouped('');
      this.jumpSections = sections;
      this.jumpOpen = true;
    });
  }

  onJumpSearch(e: any) {
    const { sections } = this.jumpRequest?.searchGrouped(e.value || '');
    this.jumpSections = sections;
  }

  onJumpSelect(e: any) {
    this.jumpRequest?.focus(e.itemData.field.id);
    this.jumpOpen = false;
  }
}
Note: The ctrlk-field-focused CSS class is added to the focused element for 2 seconds. Add this to your global styles: .ctrlk-field-focused { outline: 2px solid var(--accent); outline-offset: 2px; }

Palette Hook

hooks/useCtrlkPalette.tsReact
import { useEffect, useState, useCallback } from 'react';
import ctrlk from '@ctrlk/core';

export function useCtrlkPalette() {
  const [open, setOpen] = useState(false);
  const [request, setRequest] = useState(null);

  useEffect(() => {
    return ctrlk.onPaletteRequest((req) => {
      setRequest(req);
      setOpen(true);
    });
  }, []);

  const close = useCallback(() => setOpen(false), []);
  return { open, request, close };
}
components/CommandPalette.tsxReact
import { useState } from 'react';
import { useCtrlkPalette } from '../hooks/useCtrlkPalette';

// Use any dialog library: Radix, MUI, Headless UI, etc.
export function CommandPalette() {
  const { open, request, close } = useCtrlkPalette();
  const [query, setQuery] = useState('');
  const results = request?.search(query) || [];

  if (!open) return null;

  return (
    <div className="palette-overlay" onClick={close}>
      <div className="palette" onClick={e => e.stopPropagation()}>
        <input
          value={query}
          onChange={e => setQuery(e.target.value)}
          placeholder="Type a command..."
          autoFocus
        />
        {results.map(cmd => (
          <div
            key={cmd.id}
            className="palette-item"
            onClick={() => { request.execute(cmd.id); close(); }}>
            <span>{cmd.title}</span>
            {cmd.shortcut && <kbd>{cmd.shortcut}</kbd>}
          </div>
        ))}
      </div>
    </div>
  );
}

Grid Integration

components/DataGrid.tsxReact
import { useEffect, useRef } from 'react';
import { AgGridReact } from 'ag-grid-react';
import { AgGridAdapter } from '@ctrlk/ag-grid';
import ctrlk from '@ctrlk/core';

export function DataGrid({ rowData, columnDefs }) {
  const gridRef = useRef(null);

  useEffect(() => {
    if (!gridRef.current?.api) return;

    // Connect — returns disconnect function for cleanup
    const disconnect = ctrlk.connectGrid(
      new AgGridAdapter(gridRef.current.api)
    );

    // Register grid commands
    const unregister = ctrlk.commands.register({
      id: 'grid.export',
      title: 'Export CSV',
      shortcut: 'Ctrl+Shift+E',
      execute: () => gridRef.current.api.exportDataAsCsv(),
    });

    return () => { disconnect(); unregister(); };
  }, [gridRef.current?.api]);

  return <AgGridReact ref={gridRef} rowData={rowData} columnDefs={columnDefs} />;
}

Palette Composable

composables/useCtrlkPalette.tsVue
import { ref, computed, onMounted, onUnmounted } from 'vue';
import ctrlk from '@ctrlk/core';

export function useCtrlkPalette() {
  const open = ref(false);
  const request = ref(null);
  const query = ref('');

  const results = computed(() =>
    request.value?.search(query.value) || []
  );

  let unsub;
  onMounted(() => {
    unsub = ctrlk.onPaletteRequest((req) => {
      request.value = req;
      query.value = '';
      open.value = true;
    });
  });
  onUnmounted(() => unsub?.());

  function select(cmd) {
    request.value?.execute(cmd.id);
    open.value = false;
  }

  return { open, query, results, select, close: () => (open.value = false) };
}
CommandPalette.vueVue
<script setup>
import { useCtrlkPalette } from './composables/useCtrlkPalette';
const { open, query, results, select, close } = useCtrlkPalette();
</script>

<template>
  <Teleport to="body">
    <div v-if="open" class="palette-overlay" @click.self="close">
      <div class="palette">
        <input v-model="query" placeholder="Type a command..." autofocus />
        <div v-for="cmd in results" :key="cmd.id"
             class="palette-item"
             @click="select(cmd)">
          {{ cmd.title }}
          <kbd v-if="cmd.shortcut">{{ cmd.shortcut }}</kbd>
        </div>
      </div>
    </div>
  </Teleport>
</template>

Grid Integration

composables/useCtrlkGrid.tsVue
import { onMounted, onUnmounted } from 'vue';
import ctrlk from '@ctrlk/core';

export function useCtrlkGrid(adapter) {
  let disconnect;
  onMounted(() => { disconnect = ctrlk.connectGrid(adapter); });
  onUnmounted(() => disconnect?.());
}

Event Payloads Reference

Event HookPayload Shape
onPaletteRequest{ commands: CommandDef[], search(q): CommandDef[], execute(id): any }
onFieldJumpRequest{ fields, search(q), searchGrouped(q): { sections: [{name, fields}] }, focus(id), setSectionOrder(order), discover(), getCompleteness() }
onShortcutsRequest{ shortcuts: [{ combo, commandId, title, category }] }
onDensityChange{ level: 'compact'|'comfortable'|'spacious', previous }
onViewSaved{ name, slot, shortcut, totalSaved, maxViews, remaining, evicted }
onViewLoaded{ name }
onCommandExecuted{ id, result }
Every hook returns an unsubscribe function. Call it in your component's cleanup lifecycle (ngOnDestroy, useEffect return, onUnmounted).
